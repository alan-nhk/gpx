// 解析 GPX 檔案數據 (加入自動偵測海拔功能) function parseGPX(gpxText) { const parser = new DOMParser(); const xmlDoc = parser.parseFromString(gpxText, "text/xml"); const trkpts = xmlDoc.getElementsByTagName("trkpt"); let latlngs = []; let elevations = []; let labels = []; let hasElevationData = false; // 新增：用來記錄是否找到海拔數據 for (let i = 0; i < trkpts.length; i++) { let lat = parseFloat(trkpts[i].getAttribute("lat")); let lon = parseFloat(trkpts[i].getAttribute("lon")); // 尋找海拔標籤 let eleNode = trkpts[i].getElementsByTagName("ele")[0]; let ele = 0; // 如果找到海拔數據，就把變數設為 true，並取出數值 if (eleNode && eleNode.textContent) { ele = parseFloat(eleNode.textContent); hasElevationData = true; } latlngs.push([lat, lon]); elevations.push(ele); labels.push(i); } drawMap(latlngs); // 如果整條路線都沒有海拔數據，就跳出提示框提醒 if (!hasElevationData) { alert("溫馨提示：這個 GPX 檔案裡沒有包含海拔數據喔！" + "這通常是因為路線是手動繪製的平面路線。" + "海拔圖將會顯示為平線。"); } drawChart(labels, elevations); }
